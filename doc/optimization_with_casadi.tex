\begin{frame}
\frametitle{Example \#1 -- Linear Programming}
Farmer wants to grow \textcolor{red}{apples} and \textcolor{blue}{bananas}.
\begin{center}
\begin{tabular}{ c|c|c|c }
\hline
  & Area needed & Fertilizer needed & Profit\\
  & [$\text{m}^2/\text{ton}$] & [kg/ton] & [USD/ton] \\
  \hline
 \textcolor{red}{A} & 4000 & 60 & 700 \\
 \textcolor{blue}{B} & 3000 & 80 & 600 \\
 \hline
\end{tabular}
\end{center}
Farmer has a field of $100\;000\;\text{m}^2$ and can legally use no more than 2000 kg fertilizer.\\
\vspace{1em}
What to plant to maximize profit?
\end{frame}


\begin{frame}
 \frametitle{Example \#1 -- Formulation}
\begin{columns}
\begin{column}{0.5\textwidth}
$m_a$, $m_b$ -- mass of apple, and banana\\

Control variables, $$w = [m_a, m_b]$$

the profit, $$P(w) = 700 m_a + 600 m_b$$\\
the area used, $$A(w) = 4000 m_a + 3000 m_b$$\\
fertilizer used, $$F(w) = 60 m_a + 80 m_b$$\\
\end{column}
\begin{column}{0.5\textwidth}  %%<--- here
$$\max_w P(w) \rightarrow \min_w [-P(w)] $$
\begin{alignat*}{2}
\text{s.t.}\quad & 0 \le A(w) \le 100000\\
& 0 \le F(w) \le 2000
\end{alignat*}
Optional constraints: \\
$$0 \le m_a, m_b $$
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]
 \frametitle{Example \#1 -- Solution in CasADi}
\begin{columns}
\begin{column}{0.5\textwidth}
 \tiny
 \begin{minted}{python}
from casadi import Opti

opti = Opti()

mass_apple = opti.variable()
mass_banana = opti.variable()
opti.minimize(-700 * mass_apple - 600 * mass_banana)

area_needed = 4000 * mass_apple + 3000 * mass_banana
fertilizer_needed = 60 * mass_apple + 80 * mass_banana

opti.subject_to(mass_apple >= 0)
opti.subject_to(mass_banana >= 0)

opti.subject_to(opti.bounded(0, area_needed, 100_000))
opti.subject_to(opti.bounded(0, fertilizer_needed, 2000))

opti.set_initial(mass_apple, 5)
opti.set_initial(mass_banana, 5)

opti.solver('ipopt')

solution = opti.solve()

print(f'Mass of apples: {solution.value(mass_apple):.2f} tons')
print(f'Mass of bananas: {solution.value(mass_banana):.2f} tons')
\end{minted}
\end{column}

\begin{column}{0.4\textwidth}

 $$ P(w) = w^2 - 6w + 13 $$

 $$\min_w P(w)$$

 $$\frac{dP(w)}{dw} = 2w - 6 = 0$$

 $$ w = 3, \, P(3) = 4 $$
